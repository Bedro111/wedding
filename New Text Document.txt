<script>

    function showInvitation() {

        const input =
            document.getElementById("nameInput");

        const name =
            input.value.trim();


        if (name === "") {

            input.focus();

            input.placeholder =
                "اكتب اسمك الأول يا خويا ❤️";

            return;

        }


        // إدخال الاسم في الدعوة

        document.getElementById("userName")
            .textContent = name;


        // إخفاء شاشة الاسم

        document.getElementById("nameScreen")
            .style.display = "none";


        // إظهار الدعوة

        document.getElementById("message")
            .style.display = "block";

    }


    // Enter يفتح الدعوة

    document
        .getElementById("nameInput")
        .addEventListener(
            "keypress",
            function(event) {

                if (event.key === "Enter") {

                    showInvitation();

                }

            }
        );

</script>
